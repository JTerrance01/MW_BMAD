2. Implement the Repository Method
Next, here's the implementation of the AddJudgmentWithScoresAsync method in your SubmissionRepository. This method encapsulates the database logic and uses a transaction to maintain data integrity.

C#

// src/MixWarz.Infrastructure/Persistence/Repositories/SubmissionRepository.cs
using Microsoft.EntityFrameworkCore;
using MixWarz.Application.Common.Interfaces;
using MixWarz.Domain.Entities;
using MixWarz.Domain.Interfaces;

namespace MixWarz.Infrastructure.Persistence.Repositories;

public class SubmissionRepository : ISubmissionRepository
{
    private readonly IAppDbContext _context;

    public SubmissionRepository(IAppDbContext context)
    {
        _context = context;
    }

    // ... other existing method implementations

    public async Task AddJudgmentWithScoresAsync(SubmissionJudgement judgement, IEnumerable<CriteriaScore> scores, CancellationToken cancellationToken)
    {
        await using var transaction = await _context.Database.BeginTransactionAsync(cancellationToken);

        try
        {
            // Add and save the SubmissionJudgement to get its ID
            _context.SubmissionJudgments.Add(judgement);
            await _context.SaveChangesAsync(cancellationToken);

            // Assign the new SubmissionJudgementId to each CriteriaScore
            foreach (var score in scores)
            {
                score.SubmissionJudgementId = judgement.Id;
            }

            // Add all the CriteriaScore records
            await _context.CriteriaScores.AddRangeAsync(scores, cancellationToken);
            await _context.SaveChangesAsync(cancellationToken);

            // Commit the transaction
            await transaction.CommitAsync(cancellationToken);
        }
        catch (Exception)
        {
            // If anything fails, roll back the transaction and re-throw
            await transaction.RollbackAsync(cancellationToken);
            throw;
        }
    }
}


3. Create the Command Handler
Finally, this command handler orchestrates the process by validating the request, mapping the data, and calling the repository.

C#

// src/MixWarz.Application/Features/Submissions/Commands/SubmitJudgment/SubmitJudgmentCommandHandler.cs
using MediatR;
using Microsoft.EntityFrameworkCore;
using MixWarz.Application.Common.Interfaces;
using MixWarz.Application.Features.Submissions.Commands.SubmitJudgment;
using MixWarz.Domain.Entities;
using MixWarz.Domain.Interfaces;

namespace MixWarz.Application.Features.Submissions.Commands.SubmitJudgment;

public class SubmitJudgmentCommandHandler : IRequestHandler<SubmitJudgmentCommand, Result<int>>
{
    private readonly ISubmissionRepository _submissionRepository;
    private readonly IAppDbContext _context;

    public SubmitJudgmentCommandHandler(ISubmissionRepository submissionRepository, IAppDbContext context)
    {
        _submissionRepository = submissionRepository;
        _context = context;
    }

    public async Task<Result<int>> Handle(SubmitJudgmentCommand request, CancellationToken cancellationToken)
    {
        var submission = await _context.Submissions.FindAsync(new object[] { request.SubmissionId }, cancellationToken);
        if (submission == null)
        {
            return Result<int>.Failure($"Submission with Id {request.SubmissionId} not found.");
        }

        var existingJudgment = await _context.SubmissionJudgments
            .AnyAsync(j => j.SubmissionId == request.SubmissionId && j.JudgeId == request.JudgeId, cancellationToken);

        if (existingJudgment)
        {
            return Result<int>.Failure("You have already submitted a judgment for this submission.");
        }

        var submissionJudgement = new SubmissionJudgement
        {
            SubmissionId = request.SubmissionId,
            JudgeId = request.JudgeId,
            CompetitionId = request.CompetitionId,
            DateJudged = DateTime.UtcNow
        };

        var criteriaScores = request.Scores.Select(s => new CriteriaScore
        {
            SubmissionId = request.SubmissionId,
            JudgingCriteriaId = s.CriteriaId,
            Score = s.Score
        }).ToList();

        try
        {
            await _submissionRepository.AddJudgmentWithScoresAsync(submissionJudgement, criteriaScores, cancellationToken);
            return Result<int>.Success(submissionJudgement.Id);
        }
        catch (Exception ex)
        {
            // In a real app, you would log this exception
            return Result<int>.Failure($"An error occurred while submitting the judgment: {ex.Message}");
        }
    }
}
